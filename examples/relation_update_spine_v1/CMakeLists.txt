# Internal consumer of the linked relation core and bounded compiler adapter.
# These build targets do not define an installed SDK.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(TexoRU1Reference LANGUAGES CXX)
  option(TEXO_RU1_REFERENCE_ONLY "Build independent fixture only" ON)
endif()
if(TEXO_RU1_REFERENCE_ONLY)
  add_executable(texoRelationUpdateReference regulatory_learning.cc)
  target_compile_definitions(texoRelationUpdateReference PRIVATE TEXO_RU1_REFERENCE_ONLY=1)
  target_compile_features(texoRelationUpdateReference PRIVATE cxx_std_20)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(texoRelationUpdateReference PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
  enable_testing()
  add_test(NAME ru1_reference_fixture COMMAND texoRelationUpdateReference)
else()
  if(NOT TARGET cellerator_relation_update_core OR NOT TARGET texo_relation_update_compiler)
    message(FATAL_ERROR "RU1 requires the linked native core and compiler targets")
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(texoRelationUpdateDemo regulatory_learning.cc)
  target_compile_features(texoRelationUpdateDemo PRIVATE cxx_std_20)
  target_link_libraries(texoRelationUpdateDemo PRIVATE cellerator_relation_update_core texo_relation_update_compiler CUDA::cudart)
  add_test(NAME ru1_demo_hybrid COMMAND texoRelationUpdateDemo --route hybrid)
  add_test(NAME ru1_demo_sparse COMMAND texoRelationUpdateDemo --route sparse)
endif()
